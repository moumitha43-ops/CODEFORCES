import java.io.*;
import java.util.*;

public class codeforces16 {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int t = Integer.parseInt(br.readLine().trim());
        StringBuilder out = new StringBuilder();

        while (t-- > 0) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            long n = Long.parseLong(st.nextToken());
            int m = Integer.parseInt(st.nextToken());

            long[] a = new long[m];
            st = new StringTokenizer(br.readLine());
            for (int i = 0; i < m; i++) a[i] = Long.parseLong(st.nextToken());
            Arrays.sort(a);

            // gaps of healthy houses between consecutive infected houses
            long[] gaps = new long[m];
            for (int i = 0; i + 1 < m; i++) gaps[i] = a[i + 1] - a[i] - 1;
            gaps[m - 1] = a[0] + n - a[m - 1] - 1;   // wrap-around gap

            Arrays.sort(gaps);   // ascending, read from the end

            long saved = 0;
            long days = 0;
            for (int i = m - 1; i >= 0; i--) {
                long rem = gaps[i] - 2 * days;
                if (rem <= 0) break;                 // smaller gaps are gone too
                if (rem == 1) {
                    saved += 1;
                    days += 1;
                } else {
                    saved += rem - 1;
                    days += 2;
                }
            }

            out.append(n - saved).append('\n');
        }
        System.out.print(out);
    }
}
